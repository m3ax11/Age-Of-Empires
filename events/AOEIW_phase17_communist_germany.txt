# Germany only: the Imperial state and Reichspakt are formally dissolved.
country_event = {
    id = aoeiw_phase17_ger.3
    title = aoeiw_phase17_ger.3.t
    desc = aoeiw_phase17_ger.3.d
    picture = GFX_report_event_generic_communist_congress
    is_triggered_only = yes
    fire_only_once = yes
    trigger = { original_tag = GER }

    option = {
        name = aoeiw_phase17_ger.3.a

        if = {
            limit = { POL = { exists = yes is_subject_of = GER } }
            set_autonomy = {
                target = POL
                autonomy_state = autonomy_free
                end_wars = no
                end_civil_wars = no
            }
        }

        if = {
            limit = { is_faction_leader = yes }
            dismantle_faction = yes
        }
        else_if = {
            limit = { is_in_faction = yes }
            leave_faction = yes
        }

        remove_ideas = {
            AOEIW_GER_legacy_0
            AOEIW_GER_legacy_1
            AOEIW_GER_legacy_2
            AOEIW_GER_legacy_3
            AOEIW_GER_legacy_4
            AOEIW6_GER_rearmament_coordination_0
            AOEIW6_GER_rearmament_coordination_1
            AOEIW6_GER_rearmament_coordination_2
            AOEIW6_GER_rearmament_coordination_3
            AOEIW9_GER_spirit_mobilised_society
            AOEIW11_faction_leadership
            AOEIW17_GER_factory_unrest
            AOEIW17_GER_general_strike
            AOEIW17_GER_red_front_militias
        }
        AOEIW9_install_workers_government = yes
        set_cosmetic_tag = GER_WORKERS_FEDERATION
        set_party_name = {
            ideology = communism
            long_name = AOEIW17_GER_workers_council_congress_long
            name = AOEIW17_GER_workers_council_congress
        }
        set_popularities = {
            democratic = 12
            fascism = 3
            communism = 68
            neutrality = 10
            imperialism = 7
        }
        retire_character = AOEIW_GER_wilhelm_ii
        promote_character = { character = GER_wilhelm_pieck ideology = stalinism }
        add_ideas = {
            AOEIW17_GER_postrevolution_industry_disruption
            AOEIW17_GER_postrevolution_command_disruption
        }
        add_political_power = -100
        add_stability = -0.08
        add_war_support = 0.08
        set_country_flag = AOEIW17_GER_workers_federation_proclaimed

        if = {
            limit = { POL = { exists = yes } }
            POL = {
                remove_ideas = AOEIW11_faction_coordination
                remove_opinion_modifier = { target = GER modifier = AOEIW_reichspakt_cooperation }
                remove_opinion_modifier = { target = GER modifier = AOEIW11_diplomatic_mission }
            }
        }
        if = {
            limit = { HOL = { exists = yes } }
            HOL = {
                remove_ideas = AOEIW11_faction_coordination
                remove_opinion_modifier = { target = GER modifier = AOEIW_reichspakt_cooperation }
                remove_opinion_modifier = { target = GER modifier = AOEIW11_diplomatic_mission }
            }
        }

        if = {
            limit = { POL = { exists = yes } }
            POL = { country_event = { id = aoeiw_phase17_ger.5 days = 1 } }
        }
        if = {
            limit = { HOL = { exists = yes } }
            HOL = { country_event = { id = aoeiw_phase17_ger.6 days = 1 } }
        }
    }
}
