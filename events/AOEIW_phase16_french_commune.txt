# France only: the Congress proclaims the Commune and ends the Bonapartist state.
country_event = {
    id = aoeiw_phase16_fra.3
    title = aoeiw_phase16_fra.3.t
    desc = aoeiw_phase16_fra.3.d
    picture = GFX_report_event_generic_communist_congress
    is_triggered_only = yes
    fire_only_once = yes
    trigger = { original_tag = FRA }

    option = {
        name = aoeiw_phase16_fra.3.a

        if = {
            limit = { is_faction_leader = yes }
            dismantle_faction = yes
        }
        else_if = {
            limit = { is_in_faction = yes }
            leave_faction = yes
        }

        remove_ideas = {
            AOEIW_FRA_legacy_0
            AOEIW_FRA_legacy_1
            AOEIW_FRA_legacy_2
            AOEIW_FRA_legacy_3
            AOEIW9_FRA_spirit_mobilised_society
            AOEIW11_faction_leadership
            AOEIW16_FRA_strike_wave
            AOEIW16_FRA_revolutionary_command_crisis
        }
        AOEIW9_install_workers_government = yes
        set_cosmetic_tag = FRA_COMMUNE
        set_party_name = {
            ideology = communism
            long_name = AOEIW16_FRA_communal_congress_long
            name = AOEIW16_FRA_communal_congress
        }
        set_popularities = {
            democratic = 12
            fascism = 3
            communism = 70
            neutrality = 10
            imperialism = 5
        }
        retire_character = AOEIW_FRA_napoleon_iv
        promote_character = { character = FRA_maurice_thorez ideology = stalinism }
        add_ideas = {
            AOEIW16_FRA_postrevolution_industry_disruption
            AOEIW16_FRA_postrevolution_command_disruption
        }
        add_political_power = -100
        add_stability = -0.08
        add_war_support = 0.05
        set_country_flag = AOEIW16_FRA_commune_proclaimed

        if = {
            limit = { AUS = { exists = yes } }
            AUS = {
                remove_ideas = AOEIW11_faction_coordination
                remove_opinion_modifier = { target = FRA modifier = AOEIW_old_order_cooperation }
            }
        }
        if = {
            limit = { BEL = { exists = yes } }
            BEL = {
                remove_ideas = AOEIW11_faction_coordination
                remove_opinion_modifier = { target = FRA modifier = AOEIW_old_order_cooperation }
            }
        }

        country_event = { id = aoeiw_phase16_fra.4 days = 1 }
        if = {
            limit = { AUS = { exists = yes } }
            AUS = { country_event = { id = aoeiw_phase16_fra.5 days = 1 } }
        }
        else_if = {
            limit = { BEL = { exists = yes } }
            BEL = { country_event = { id = aoeiw_phase16_fra.6 days = 1 } }
        }
    }
}
