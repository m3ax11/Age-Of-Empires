# Italy only: abolish the monarchy and dissolve the Mediterranean League.
country_event = {
    id = aoeiw_phase18_ita.3
    title = aoeiw_phase18_ita.3.t
    desc = aoeiw_phase18_ita.3.d
    picture = GFX_report_event_generic_communist_congress
    is_triggered_only = yes
    fire_only_once = yes
    trigger = { original_tag = ITA }

    option = {
        name = aoeiw_phase18_ita.3.a

        if = {
            limit = { is_faction_leader = yes }
            dismantle_faction = yes
        }
        else_if = {
            limit = { is_in_faction = yes }
            leave_faction = yes
        }

        remove_ideas = {
            AOEIW_ITA_legacy_0
            AOEIW_ITA_legacy_1
            AOEIW_ITA_legacy_2
            AOEIW_ITA_legacy_3
            AOEIW9_ITA_spirit_mobilised_society
            AOEIW11_faction_leadership
            AOEIW18_ITA_factory_council_agitation
            AOEIW18_ITA_po_valley_general_strike
            AOEIW18_ITA_red_guard_militias
        }
        AOEIW9_install_workers_government = yes
        set_cosmetic_tag = ITA_COMMUNES
        set_party_name = {
            ideology = communism
            long_name = AOEIW18_ITA_council_congress_long
            name = AOEIW18_ITA_council_congress
        }
        set_popularities = {
            democratic = 10
            fascism = 5
            communism = 70
            neutrality = 8
            imperialism = 7
        }
        retire_character = ITA_vittorio_emanuele_iii
        promote_character = { character = ITA_palmiro_togliatti ideology = marxism }
        add_ideas = {
            AOEIW18_ITA_postrevolution_industry_disruption
            AOEIW18_ITA_postrevolution_command_disruption
        }
        add_political_power = -100
        add_stability = -0.08
        add_war_support = 0.08
        set_country_flag = AOEIW18_ITA_federation_proclaimed

        if = {
            limit = { ROM = { exists = yes } }
            ROM = {
                remove_ideas = AOEIW11_faction_coordination
                remove_opinion_modifier = { target = ITA modifier = AOEIW_mediterranean_league_cooperation }
                remove_opinion_modifier = { target = ITA modifier = AOEIW11_diplomatic_mission }
                country_event = { id = aoeiw_phase18_ita.5 days = 1 }
            }
        }
        if = {
            limit = { YUG = { exists = yes } }
            YUG = {
                remove_ideas = AOEIW11_faction_coordination
                remove_opinion_modifier = { target = ITA modifier = AOEIW_mediterranean_league_cooperation }
                remove_opinion_modifier = { target = ITA modifier = AOEIW11_diplomatic_mission }
                country_event = { id = aoeiw_phase18_ita.6 days = 2 }
            }
        }
        if = {
            limit = { BUL = { exists = yes } }
            BUL = {
                remove_ideas = AOEIW11_faction_coordination
                remove_opinion_modifier = { target = ITA modifier = AOEIW_mediterranean_league_cooperation }
                remove_opinion_modifier = { target = ITA modifier = AOEIW11_diplomatic_mission }
                country_event = { id = aoeiw_phase18_ita.7 days = 3 }
            }
        }
    }
}
