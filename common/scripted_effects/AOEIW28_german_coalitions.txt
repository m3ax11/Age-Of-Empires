    set_country_flag = AOEIW28_GER_parliament_enabled
    clr_country_flag = AOEIW30_GER_bundestag
    clr_country_flag = AOEIW30_GER_reichstag
    if = {
        limit = { has_government = democratic }
        set_country_flag = AOEIW30_GER_bundestag
        if = { limit = { NOT = { has_country_flag = AOEIW34_GER_polls_initialized } } AOEIW34_GER_seed_bundestag_polls = yes }
    }
    else = {
        set_country_flag = AOEIW30_GER_reichstag
        if = { limit = { NOT = { has_country_flag = AOEIW34_GER_polls_initialized } } AOEIW34_GER_seed_reichstag_polls = yes }
    }
    set_country_flag = AOEIW34_GER_polls_initialized
    mark_focus_tree_layout_dirty = yes
}
