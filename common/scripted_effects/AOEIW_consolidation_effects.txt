AOEIW_consolidation_apply_political_settlement = {
    if = {
        limit = { NOT = { has_country_flag = AOEIW_consolidation_political_complete } }
        set_country_flag = AOEIW_consolidation_political_complete
        add_political_power = 75
        add_stability = 0.04
        add_ideas = AOEIW_consolidation_political_settlement
    }
}

AOEIW_consolidation_apply_industrial_programme = {
    if = {
        limit = { NOT = { has_country_flag = AOEIW_consolidation_economic_complete } }
        set_country_flag = AOEIW_consolidation_economic_complete
        add_ideas = AOEIW_consolidation_industrial_programme
        add_tech_bonus = { name = AOEIW_consolidation_industry_bonus bonus = 0.50 uses = 2 category = industry }
        random_owned_controlled_state = {
            limit = { is_core_of = ROOT }
            add_extra_state_shared_building_slots = 1
            add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
        }
    }
}

AOEIW_consolidation_apply_military_reform = {
    if = {
        limit = { NOT = { has_country_flag = AOEIW_consolidation_military_complete } }
        set_country_flag = AOEIW_consolidation_military_complete
        add_ideas = AOEIW_consolidation_military_reform
        army_experience = 20
        air_experience = 10
        navy_experience = 10
        add_doctrine_cost_reduction = { name = AOEIW_consolidation_doctrine_bonus cost_reduction = 0.25 uses = 1 category = land_doctrine }
    }
}

AOEIW_consolidation_apply_diplomatic_charter = {
    if = {
        limit = { NOT = { has_country_flag = AOEIW_consolidation_diplomatic_complete } }
        set_country_flag = AOEIW_consolidation_diplomatic_complete
        add_ideas = AOEIW_consolidation_diplomatic_charter
        add_political_power = 75
    }
}
