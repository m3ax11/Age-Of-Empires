# AOEIW V10 PHASE 8 — Portuguese AI mobilisation
# Threat-staged force development for a small Atlantic empire. Only one stage is active at a time.
# Values are ordinary AI desires/production minima; no free manpower/equipment or hidden output bonuses.

AOEIW_POR_peacetime_posture = {
    allowed = { original_tag = POR }
    enable = {
        is_ai = yes
        has_war = no
        threat < 0.25
        NOT = { any_claim = yes }
        NOT = { has_wargoal_against = SPR }
        NOT = {
            any_other_country = { has_wargoal_against = ROOT }
        }
    }
    abort_when_not_enabled = yes

    ai_strategy = { type = added_military_to_civilian_factory_ratio value = 0 }
    ai_strategy = { type = build_army id = infantry value = 18 }
    ai_strategy = { type = build_army id = AOEIW_mobile value = 1 }
    ai_strategy = { type = equipment_stockpile_surplus_ratio value = 25 }

}
