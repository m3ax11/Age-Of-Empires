scripted_map_modes = {
    imperial_world_geopolitics_map_mode = {
        top = {
            type = country
            color = {
                FROM = {
                    set_temp_variable = { alpha = 0.96 }
                    set_temp_variable = { thickness = 3.0 }

                    # The Chinese national war is deliberately legible at a glance.
                    if = {
                        limit = {
                            OR = {
                                has_country_flag = AOEIW20_UF_war_started
                                has_country_flag = AOEIW20_UF_nra_path
                                has_country_flag = AOEIW20_UF_mao_path
                            }
                        }
                        set_temp_variable = { red = 0.49 }
                        set_temp_variable = { green = 0.12 }
                        set_temp_variable = { blue = 0.16 }
                    }
                    # Japan and its Manchurian client form the opposing continental bloc.
                    else_if = {
                        limit = {
                            OR = {
                                original_tag = JAP
                                original_tag = MAN
                                is_in_faction_with = JAP
                            }
                        }
                        set_temp_variable = { red = 0.55 }
                        set_temp_variable = { green = 0.43 }
                        set_temp_variable = { blue = 0.19 }
                    }

                    # The Old Order: royal blue and cold steel.
                    else_if = {
                        limit = { is_in_faction_with = FRA }
                        set_temp_variable = { red = 0.16 }
                        set_temp_variable = { green = 0.27 }
                        set_temp_variable = { blue = 0.45 }
                    }
                    # Reichspakt: Prussian charcoal-blue.
                    else_if = {
                        limit = { is_in_faction_with = GER }
                        set_temp_variable = { red = 0.10 }
                        set_temp_variable = { green = 0.15 }
                        set_temp_variable = { blue = 0.22 }
                    }
                    # Mediterranean League: restrained copper and ochre.
                    else_if = {
                        limit = { is_in_faction_with = ITA }
                        set_temp_variable = { red = 0.43 }
                        set_temp_variable = { green = 0.27 }
                        set_temp_variable = { blue = 0.12 }
                    }
                    # Revolutionary governments and their aligned blocs.
                    else_if = {
                        limit = {
                            OR = {
                                has_government = communism
                                is_in_faction_with = SOV
                            }
                        }
                        set_temp_variable = { red = 0.49 }
                        set_temp_variable = { green = 0.10 }
                        set_temp_variable = { blue = 0.13 }
                    }
                    # Imperial states outside the three principal blocs.
                    else_if = {
                        limit = { has_government = imperialism }
                        set_temp_variable = { red = 0.48 }
                        set_temp_variable = { green = 0.38 }
                        set_temp_variable = { blue = 0.16 }
                    }
                    # Parliamentary states outside the principal blocs.
                    else_if = {
                        limit = { has_government = democratic }
                        set_temp_variable = { red = 0.18 }
                        set_temp_variable = { green = 0.39 }
                        set_temp_variable = { blue = 0.39 }
                    }
                    else = {
                        set_temp_variable = { red = 0.29 }
                        set_temp_variable = { green = 0.31 }
                        set_temp_variable = { blue = 0.34 }
                    }

                    # Subjects remain legible but visibly subordinate.
                    if = {
                        limit = { is_subject = yes }
                        set_temp_variable = { alpha = 0.68 }
                    }
                    # Active belligerents receive a restrained bright border.
                    if = {
                        limit = { has_war = yes }
                        set_temp_variable = { thickness = 5.0 }
                        set_temp_variable = { highlighted = 1.0 }
                    }

                    # Scripted map-mode buttons are registered while the game is
                    # loading, so the game rule disables the custom presentation
                    # by replacing it with a single unobtrusive neutral layer.
                    if = {
                        limit = {
                            has_game_rule = {
                                rule = AOEIW_imperial_world_map_mode_rule
                                option = AOEIW_MAP_MODE_DISABLED
                            }
                        }
                        set_temp_variable = { red = 0.27 }
                        set_temp_variable = { green = 0.27 }
                        set_temp_variable = { blue = 0.27 }
                        set_temp_variable = { alpha = 0.45 }
                        set_temp_variable = { thickness = 1.0 }
                        set_temp_variable = { highlighted = 0.0 }
                    }
                }
            }
            thickness = yes
        }

        bottom = {
            type = game_map_mode_country
        }

        far_text = faction
        near_text = country
        update_daily = yes
    }
}
